cmake_minimum_required(VERSION 3.22.1)
project(pocket_phonemizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_program(NODE_EXECUTABLE node REQUIRED)
execute_process(COMMAND "${CMAKE_COMMAND}" -E env "CMAKE=${CMAKE_COMMAND}" "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/prepare-source.mjs" RESULT_VARIABLE PREPARE_RESULT)
if(NOT PREPARE_RESULT EQUAL 0)
  message(FATAL_ERROR "Pinned offline phonemizer source verification failed")
endif()
execute_process(COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify-data.mjs" RESULT_VARIABLE DATA_RESULT)
if(NOT DATA_RESULT EQUAL 0)
  message(FATAL_ERROR "Pinned offline phonemizer data verification failed")
endif()
set(ESPEAK_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../generated/source/espeak-ng")
if(NOT EXISTS "${ESPEAK_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/../cpp/data-identity.h")
  message(FATAL_ERROR "Prepare the pinned offline phonemizer source and data before native configuration")
endif()
foreach(option BUILD_SHARED_LIBS COMPILE_INTONATIONS ENABLE_TESTS BUILD_TESTING USE_ASYNC USE_MBROLA USE_LIBSONIC USE_LIBPCAUDIO USE_KLATT USE_SPEECHPLAYER)
  set(${option} OFF CACHE BOOL "Offline phoneme-only module" FORCE)
endforeach()
add_subdirectory("${ESPEAK_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/espeak" EXCLUDE_FROM_ALL)
add_library(pocket_phonemizer SHARED ../cpp/PhonemizerCore.cpp ../cpp/PhonemizerJni.cpp)
target_link_libraries(pocket_phonemizer PRIVATE espeak-ng)
target_link_options(pocket_phonemizer PRIVATE "-Wl,-z,max-page-size=16384")
