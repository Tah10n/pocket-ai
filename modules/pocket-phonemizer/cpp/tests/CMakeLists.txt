cmake_minimum_required(VERSION 3.22)
project(pocket_phonemizer_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
get_filename_component(MODULE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ESPEAK_SOURCE "${MODULE_ROOT}/generated/source/espeak-ng")
if(NOT EXISTS "${ESPEAK_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Prepare the pinned offline source before configuring native tests")
endif()
file(TO_CMAKE_PATH "$ENV{JAVA_HOME}" TEST_JAVA_HOME)
if(NOT EXISTS "${TEST_JAVA_HOME}/include/jni.h")
  message(FATAL_ERROR "Native boundary tests require JAVA_HOME pointing to a JDK with JNI headers")
endif()
if(WIN32)
  set(JNI_PLATFORM win32)
elseif(APPLE)
  set(JNI_PLATFORM darwin)
else()
  set(JNI_PLATFORM linux)
endif()
if(NOT EXISTS "${TEST_JAVA_HOME}/include/${JNI_PLATFORM}/jni_md.h")
  message(FATAL_ERROR "JAVA_HOME lacks platform JNI headers")
endif()
foreach(option BUILD_SHARED_LIBS COMPILE_INTONATIONS ENABLE_TESTS BUILD_TESTING USE_ASYNC USE_MBROLA USE_LIBSONIC USE_LIBPCAUDIO USE_KLATT USE_SPEECHPLAYER)
  set(${option} OFF CACHE BOOL "Offline host phoneme tests" FORCE)
endforeach()
add_subdirectory("${ESPEAK_SOURCE}" "${CMAKE_BINARY_DIR}/espeak" EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_executable(core_lifecycle ../PhonemizerCore.cpp core-lifecycle.cpp)
add_executable(core_real ../PhonemizerCore.cpp core-real.cpp)
add_executable(jni_failures jni-failures.cpp)
target_link_libraries(core_lifecycle PRIVATE Threads::Threads)
target_link_libraries(core_real PRIVATE espeak-ng Threads::Threads)
if(WIN32)
  target_link_libraries(core_real PRIVATE advapi32)
endif()
foreach(target core_lifecycle core_real)
  target_include_directories(${target} PRIVATE "${ESPEAK_SOURCE}/src/include")
  if(WIN32)
    target_compile_definitions(${target} PRIVATE LIBESPEAK_NG_EXPORT=1)
  endif()
endforeach()
target_include_directories(jni_failures PRIVATE "${TEST_JAVA_HOME}/include" "${TEST_JAVA_HOME}/include/${JNI_PLATFORM}")
foreach(target core_lifecycle core_real jni_failures)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()
